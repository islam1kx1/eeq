-- Drop table

-- DROP TABLE public."Dealers";

CREATE TABLE public."Dealers" (
	dealerid serial4 NOT NULL,
	"name" varchar(255) NULL,
	region varchar(255) NULL,
	phone varchar(255) NULL,
	contactperson varchar(255) NULL,
	CONSTRAINT "Dealers_pkey" PRIMARY KEY (dealerid)
);


-- public."ModelLines" определение

-- Drop table

-- DROP TABLE public."ModelLines";

CREATE TABLE public."ModelLines" (
	modelid serial4 NOT NULL,
	"name" varchar(255) NULL,
	"type" varchar(100) NULL,
	specifications text NULL,
	price numeric(12, 2) NULL,
	CONSTRAINT "ModelLines_pkey" PRIMARY KEY (modelid)
);


-- public."Shipments" определение

-- Drop table

-- DROP TABLE public."Shipments";

CREATE TABLE public."Shipments" (
	shipmentid serial4 NOT NULL,
	modelid int4 NULL,
	dealerid int4 NULL,
	quantity int4 NULL,
	"date" date NULL,
	CONSTRAINT "Shipments_pkey" PRIMARY KEY (shipmentid),
	CONSTRAINT "Shipments_dealerid_fkey" FOREIGN KEY (dealerid) REFERENCES public."Dealers"(dealerid) ON DELETE CASCADE,
	CONSTRAINT "Shipments_modelid_fkey" FOREIGN KEY (modelid) REFERENCES public."ModelLines"(modelid) ON DELETE CASCADE
);
